# PLAYR Angular Auth and Profile Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first Angular frontend for PLAYR with register, login, protected current-user profile, public profile, and profile editing against the existing ASP.NET Core API.

**Architecture:** Add a standalone Angular app under `src/Playr.Web`. Keep app-wide API/auth/profile services under `src/app/core`, feature pages under `src/app/features`, and routing in `app.routes.ts`. Use JWT in `localStorage`, a functional HTTP interceptor, and a functional route guard.

**Tech Stack:** Angular CLI latest stable, standalone Angular components, TypeScript, Angular Router, Angular HttpClient, CSS, existing ASP.NET Core REST API.

## Global Constraints

- Frontend framework: modern Angular with standalone components.
- Frontend location: `src/Playr.Web`.
- API integration: REST calls to the existing ASP.NET Core API.
- Authentication: JWT stored in `localStorage` for the MVP.
- Route protection: functional Angular route guards.
- HTTP authentication: functional Angular HTTP interceptor registered with `provideHttpClient(withInterceptors(...))`.
- Styling: simple custom CSS, responsive enough for desktop and mobile.
- Do not implement Games UI, Feed UI, Posts/play logs UI, Looking-to-play UI, Threads UI, file uploads, refresh tokens, or a complex design system.
- Do not commit unless the user explicitly asks for commits.

- [ ] **Step 5: Add auth service**

Create `src/Playr.Web/src/app/core/auth/auth.service.ts`:

```typescript
import { HttpClient } from '@angular/common/http';
import { Injectable, signal } from '@angular/core';
import { Observable, tap } from 'rxjs';

import { API_BASE_URL } from '../api/api.config';
import { AuthUser, LoginRequest, LoginResponse, RegisterRequest } from './auth.models';

const TOKEN_KEY = 'playr_access_token';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly tokenState = signal<string | null>(localStorage.getItem(TOKEN_KEY));
  private readonly currentUserState = signal<AuthUser | null>(null);

  readonly token = this.tokenState.asReadonly();
  readonly currentUser = this.currentUserState.asReadonly();

  constructor(private readonly http: HttpClient) {}

  register(request: RegisterRequest): Observable<AuthUser> {
    return this.http.post<AuthUser>(`${API_BASE_URL}/auth/register`, request);
  }

  login(request: LoginRequest): Observable<LoginResponse> {
    return this.http.post<LoginResponse>(`${API_BASE_URL}/auth/login`, request).pipe(
      tap((response) => this.setToken(response.accessToken)),
    );
  }

  loadCurrentUser(): Observable<AuthUser> {
    return this.http.get<AuthUser>(`${API_BASE_URL}/auth/me`).pipe(
      tap((user) => this.currentUserState.set(user)),
    );
  }

  logout(): void {
    localStorage.removeItem(TOKEN_KEY);
    this.tokenState.set(null);
    this.currentUserState.set(null);
  }

  private setToken(token: string): void {
    localStorage.setItem(TOKEN_KEY, token);
    this.tokenState.set(token);
  }
}
```

```typescript
import { HttpHandlerFn, HttpInterceptorFn, HttpRequest } from '@angular/common/http';
import { inject } from '@angular/core';

import { AuthService } from './auth.service';

export const authInterceptor: HttpInterceptorFn = (request: HttpRequest<unknown>, next: HttpHandlerFn) => {
  const token = inject(AuthService).token();
  if (!token) {
    return next(request);
  }

  return next(request.clone({ setHeaders: { Authorization: `Bearer ${token}` } }));
};
```

```typescript
import { Component, inject, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { Router, RouterLink } from '@angular/router';

import { AuthService } from '../../../core/auth/auth.service';

@Component({
  selector: 'app-login-page',
  imports: [FormsModule, RouterLink],
  templateUrl: './login-page.component.html',
  styleUrl: './login-page.component.css',
})
export class LoginPageComponent {
  protected usernameOrEmail = '';
  protected password = '';
  protected readonly isSubmitting = signal(false);
  protected readonly error = signal<string | null>(null);

  private readonly auth = inject(AuthService);
  private readonly router = inject(Router);

  protected submit(): void {
    this.error.set(null);
    this.isSubmitting.set(true);

    this.auth.login({ usernameOrEmail: this.usernameOrEmail, password: this.password }).subscribe({
      next: () => {
        this.isSubmitting.set(false);
        void this.router.navigateByUrl('/me');
      },
      error: () => {
        this.isSubmitting.set(false);
        this.error.set('Login failed. Check your username/email and password.');
      },
    });
  }
}
```

import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';

import { API_BASE_URL } from '../api/api.config';
import { Profile, UpdateProfileRequest } from './profile.models';

@Injectable({ providedIn: 'root' })
export class ProfileService {
  constructor(private readonly http: HttpClient) {}

  getByUsername(username: string): Observable<Profile> {
    return this.http.get<Profile>(`${API_BASE_URL}/profiles/${encodeURIComponent(username)}`);
  }

  updateMe(request: UpdateProfileRequest): Observable<Profile> {
    return this.http.put<Profile>(`${API_BASE_URL}/profiles/me`, request);
  }
}

import { Component, inject, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { Router } from '@angular/router';

import { AuthService } from '../../../core/auth/auth.service';
import { UpdateProfileRequest } from '../../../core/profile/profile.models';
import { ProfileService } from '../../../core/profile/profile.service';

@Component({
  selector: 'app-edit-profile-page',
  imports: [FormsModule],
  templateUrl: './edit-profile-page.component.html',
  styleUrl: './edit-profile-page.component.css',
})
export class EditProfilePageComponent {
  protected displayName = '';
  protected bio = '';
  protected avatarUrl = '';
  protected region = '';
  protected languages = '';
  protected platforms = '';
  protected externalLinks = '';
  protected currentlyPlayingGames = '';
  protected lookingForPlayers = false;
  protected readonly error = signal<string | null>(null);
  protected readonly isSubmitting = signal(false);

  private readonly auth = inject(AuthService);
  private readonly profiles = inject(ProfileService);
  private readonly router = inject(Router);

  constructor() {
    this.auth.loadCurrentUser().subscribe({
      next: (user) => this.loadProfile(user.username),
      error: () => this.error.set('Could not load your user session.'),
    });
  }

  protected submit(): void {
    this.error.set(null);
    this.isSubmitting.set(true);
    this.profiles.updateMe(this.toUpdateRequest()).subscribe({
      next: () => {
        this.isSubmitting.set(false);
        void this.router.navigateByUrl('/me');
      },
      error: () => {
        this.isSubmitting.set(false);
        this.error.set('Could not update your profile. Check URLs and field lengths.');
      },
    });
  }

  private loadProfile(username: string): void {
    this.profiles.getByUsername(username).subscribe({
      next: (profile) => {
        this.displayName = profile.displayName;
        this.bio = profile.bio ?? '';
        this.avatarUrl = profile.avatarUrl ?? '';
        this.region = profile.region ?? '';
        this.languages = profile.languages.join('\n');
        this.platforms = profile.platforms.join('\n');
        this.externalLinks = Object.entries(profile.externalLinks).map(([key, value]) => `${key}=${value}`).join('\n');
        this.currentlyPlayingGames = profile.currentlyPlayingGames.join('\n');
        this.lookingForPlayers = profile.lookingForPlayers;
      },
      error: () => this.error.set('Could not load your profile.'),
    });
  }

  private toUpdateRequest(): UpdateProfileRequest {
    return {
      displayName: this.displayName,
      bio: this.emptyToNull(this.bio),
      avatarUrl: this.emptyToNull(this.avatarUrl),
      region: this.emptyToNull(this.region),
      languages: this.toList(this.languages),
      platforms: this.toList(this.platforms),
      externalLinks: this.toExternalLinks(this.externalLinks),
      currentlyPlayingGames: this.toList(this.currentlyPlayingGames),
      lookingForPlayers: this.lookingForPlayers,
    };
  }

  private toList(value: string): string[] {
    return value.split('\n').map((item) => item.trim()).filter(Boolean);
  }

  private toExternalLinks(value: string): Record<string, string> {
    return value.split('\n').map((line) => line.trim()).filter(Boolean).reduce<Record<string, string>>((links, line) => {
      const [key, ...rest] = line.split('=');
      const normalizedKey = key.trim();
      const normalizedValue = rest.join('=').trim();
      if (normalizedKey && normalizedValue) {
        links[normalizedKey] = normalizedValue;
      }
      return links;
    }, {});
  }

  private emptyToNull(value: string): string | null {
    const trimmed = value.trim();
    return trimmed.length === 0 ? null : trimmed;
  }
}
